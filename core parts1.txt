            r"[ㅠㅜ]+[ㅋㅎ]+",                # ㅠㅠㅋ, ㅜㅜㅎ 등

    def _irony_context_window(self, text: str, window_size: int = 15) -> float:
        """문맥 윈도우 내 반어 강도 계산"""
        words = text.split()
        irony_score = 0.0
        
        for i, word in enumerate(words):
            if any(re.search(p, word) for p in self.irony_patterns):
                # 앞뒤 window_size 내에서 부정어 검색
                start = max(0, i - window_size)
                end = min(len(words), i + window_size + 1)
                context = " ".join(words[start:end])
                
                neg_count = sum(1 for neg in self.negative_core if neg in context)
                if neg_count > 0:
                    irony_score += min(0.15 * neg_count, 0.45)
                    
        return float(np.clip(irony_score, 0.0, 0.6))

    def _apply_irony_laughter(self, e: EmotionScore, text: str):
        """개선된 반어 처리"""
        base_irony = 0.0

### Branch-B: Dirichlet 하한선 및 TV 정규화 가드

```python
# emotion_score.py 패치
@dataclass
class EmotionScore:
    # 기존 필드들...
    
    def normalize(self, dirichlet_alpha: float = 0.01, tv_lambda: float = 0.05):
        """
        Dirichlet 하한선 + Total Variation 정규화 적용
        """
        v = np.clip(self.to_vec(), 0.0, 1.0)
        nn = v[:-1]  # neutral 제외
        
        # Dirichlet 하한선 (모든 성분에 최소값 보장)
        nn = nn + dirichlet_alpha
        
        s = float(nn.sum())
        if s > 0:
            nn = nn / s
            
            # Total Variation 정규화 (과도한 스파이크 억제)
            if tv_lambda > 0:
                sorted_indices = np.argsort(nn)[::-1]
                for i in range(len(sorted_indices) - 1):
                    curr_idx = sorted_indices[i]
                    next_idx = sorted_indices[i + 1]
                    diff = nn[curr_idx] - nn[next_idx]
                    if diff > 0.7:  # 임계값 이상 차이
                        transfer = min(diff * tv_lambda, diff * 0.3)
                        nn[curr_idx] -= transfer
                        nn[next_idx] += transfer
            
            # 재정규화
            nn = nn / nn.sum()
            v[:-1] = nn
            v[-1] = 0.0
        else:
            v = np.zeros_like(v)
            v[-1] = 1.0
            
        self.from_vec(v)

    def enforce_top2_threshold(self, theta: float = 0.2):
        """top-2 합계 임계값 강제"""
        v = self.to_vec()[:-1]  # neutral 제외
        if len(v) < 2:
            return
            
        sorted_indices = np.argsort(v)[::-1]
        top2_sum = v[sorted_indices[0]] + v[sorted_indices[1]]
        
        if top2_sum < theta:
            # 부족한 만큼 top-2에 균등 분배
            deficit = theta - top2_sum
            v[sorted_indices[0]] += deficit * 0.6
            v[sorted_indices[1]] += deficit * 0.4
            
            # 정규화
            v = v / v.sum()
            vnew = np.zeros(12)
            vnew[:-1] = v
            self.from_vec(vnew)
```

- Core Parts No. 4 GROK4 -

GROK-4: 곡선 실험 보고서 — 벨 다항 분배 및 부등식 회수 캡

COSMOS 시스템의 완성된 버전(Branch-A RuleGraph와 Branch-B ProbFlow 통합 앙상블)을 가정하면, 이 실험은 ESTimeline 내 phrase 경계에서의 intensity/tension 곡선 분배를 최적화하는 핵심 모듈로 통합됩니다. RuleGraph에서는 결정론적 가드(예: top-2 라벨 합 ≥0.2)를 통해 과도 피크를 사전 필터링하고, ProbFlow에서는 에너지 확산(resonance_multi 확률 스코어링)을 통해 부드러운 회수를 적용합니다. 공통 API의 `analyze()` 호출 시, fps=25 기준으로 phrase 길이 N(프레임 수)에 따라 동적으로 k(곡률)와 캡계수(cap_coeff)를 선택하여 tension∩곡선과 cadence→rhythm 매핑을 강화합니다. 전체 시스템 예측: 10k 프레임 <150ms 처리(앙상블 폴백 포함), diagnostics에 평활도/피크 보존율 추가로 neutral 제외 위반 0% 달성. 테스트 코퍼스(예: "아 진짜 짜증나네...ㅎ")에서 짜증(anger 0.6 + han 0.3) 피크가 p=0.4 위치에서 resonance_phase=0.85로 공명하며, MusicParameters의 dissonance_rate=0.3으로 매핑됩니다.

실험 설계: 
- **입력 범위**: Phrase 길이 N (short: N≤50, long: N>50 예시로 20/100 사용), 피크 위치 p∈[0.3,0.7] (감정 피크가 phrase 초/중/후에 치우침 반영), 곡률 k∈[0.5,3.0] (낮을수록 부드러움, 높을수록 급변 재현).

### Analysis of Hierarchical Structure

The COSMOS system is architected for maximal expressivity and mimicry of complex emotional dynamics seen in natural or synthetic agents. Moving further down, the **Systemic (Macro) Layer** encapsulates large functional regions—such as cognitive modules, affect orchestration engines, or environmental processors. **Bidirectional propagation** is a foundational principle: emotion signals and regulation can travel *downward* (top-down: context setting, restraint, or goal alignment) and *upward* (bottom-up: aggregation, emergent feedback, anomaly signaling). Lateral influence is possible within layers, fostering self-organization and redundancy. Each propagation path is subject to modulation through "cosmic velocity" settings, which dynamically adjust the timing and intensity of emotional information transfer across the hierarchy, supporting both rapid reactivity and prolonged, smoothed transitions[43dcd9a7-70db-4a1f-b0ae-981daa162054](https://github.com/IdeasCosmos/COSMOS-System?citationMarker=43dcd9a7-70db-4a1f-b0ae-981daa162054 "3").

### Bidirectional Propagation Rules (Up↔Down)

In COSMOS, emotional signals do not merely cascade from a single point of origin. Instead, the propagation paradigm operates under a *bidirectional* scheme, recognizing the importance of both **contextual constraint** (top-down) and **emergent aggregation** (bottom-up).

- **Downward propagation** typically initiates with higher-layer intent—such as a scenario shift, goal priming, or regulatory override. For example, if the Universal layer detects environmental instability, it can dampen arousal parameters across all subordinate nodes, effectively imposing a global "calm" state.
- **Upward propagation** allows nodes or local subsystems to signal strong affective states, trigger system-wide alerts, or demand resource reallocation. A dramatic increase in dissonance at the node level, for example, can propagate upward recursively, triggering recalibration or even a context reconsideration at the Universal layer[43dcd9a7-70db-4a1f-b0ae-981daa162054](https://github.com/IdeasCosmos/Cosmos_emotional-vector/blob/main/cosmos_architecture.py?citationMarker=43dcd9a7-70db-4a1f-b0ae-981daa162054 "2")[43dcd9a7-70db-4a1f-b0ae-981daa162054](https://github.com/IdeasCosmos/COSMOS-System?citationMarker=43dcd9a7-70db-4a1f-b0ae-981daa162054 "3").
- The interplay is often tightly regulated to *prevent oscillatory instability* or feedback loops that could lock the system into undesirable emotional attractor states.

The *rules* for propagation prioritize *weighted averaging*, *time-windowed smoothing*, and, when necessary, *priority override* flags. Critically, each step along the propagation path references cosmic velocity constraints, ensuring that neither upwelling nor top-down responses disrupt system harmony unless so designed.

### Cosmic Velocity Regulation

A crucial innovation in the COSMOS architecture is the introduction of the **cosmic velocity** parameter. This scalar regulates the *speed* (rate of transmission) and, in advanced implementations, the *directional bias* (favoring up or down flow) of emotional propagation between layers and nodes.

- At **low cosmic velocities**, changes diffuse gradually, resulting in slow, continuous emotional transitions—ideal for background state regulation or for environments that demand stability over sensationalism.
- At **high cosmic velocities**, emotional states can propagate rapidly, allowing for strong reactivity to stimuli or for dramatic global mood shifts. This is beneficial in contexts where rapid adaptation is necessary, such as safety-critical systems or real-time interactive agents[43dcd9a7-70db-4a1f-b0ae-981daa162054](https://scisimple.com/en/articles/2025-05-23-understanding-cosmic-velocities-and-their-significance--ak4ojvw?citationMarker=43dcd9a7-70db-4a1f-b0ae-981daa162054 "4").

By calibrating cosmic velocity, COSMOS maintainers can strike a balance between agility and steadiness, optimizing for specific domain requirements (e.g., ambient emotion for creative AI, safety compliance for autonomous systems). Moreover, the cosmic velocity parameter is often exposed as a tunable variable within system APIs, with boundary checks enforced to prevent destabilizing feedback.

In summary, the *combination* of hierarchical structuring, bidirectional rules, and finely-adjusted cosmic velocity enables COSMOS to support intricate affective modeling, avoiding both sluggishness and the perils of hypersensitive overreaction.

**Dissonance Rate** in COSMOS represents the aggregate measure of system "unrest" or instability, operationalized through several discrete yet interacting components—as summarized above. *Pitch deviation* calculates the departure of current pitch content from a defined tonal center, with larger deviations raising dissonance. *Temporal jitter* reflects timing irregularities (either intentional, for expressive purposes, or inadvertent, denoting loss of coherence), and contributes to perceived instability. *Amplitude variance* encodes fluctuations in signal or node strength, where excessive variance is often a sign of agitation or conflict.

One of the more sophisticated components is **harmonic tension**, which quantifies the degree to which musical or expressive intervals remain unresolved. For example, stacking minor seconds or tritones (musically dissonant intervals) will sharply raise the computed dissonance rate. *Phase conflict* addresses synchrony at a higher-order level—nodes or modular oscillators falling "out of phase" boosts overall dissonance, a dynamic especially relevant when simulating chaotic or emergent social situations in affective agents. Finally, *contextual clashes* represent mismatches between node or module output and the current global emotional context, provoking cognitive dissonance or mood instability at the system level.

By weighting and aggregating these components, COSMOS can produce a single dissonance_rate scalar that is contextually aware and dynamically sensitive, underpinning much of the system's adaptive and regulatory capacity.

## Guide to Key Tuning Parameters in COSMOS Emotion Scoring

### Overview

The performance and expressive accuracy of the COSMOS emotion scoring system depend heavily on three core tuning parameters: **up_rate**, **down_decay**, and the **top-k guard slider**. These parameters govern, respectively, the rate of upward signal propagation, the attenuation of signals moving downward, and the strategic filtering of most salient emotional vectors. A thorough understanding of their design, effects, and recommended usage is crucial for precise emotional modeling and regulation in diverse application contexts.

---

### Parameter 1: up_rate

**Role**

The **up_rate** parameter controls the *amplification and transmission speed of bottom-up signals* within the COSMOS hierarchy. When a local node or subsystem detects a significant affective event (e.g., a spike in arousal, dissonance, or contextually salient change), up_rate determines how forcefully and rapidly this local change is broadcast upwards through the module, systemic, and ultimately to the global context layers.

**Effects on System Dynamics**

- **High up_rate:**
  - Local changes exert a *strong, immediate influence* over higher-level states. Sudden node-level disruptions (e.g., local alarm triggers) can almost instantly propagate through the system, driving global recalibration or state shifts.
  - Enhances the system's *responsiveness*, making it highly adaptive to emergent or unexpected events.
  - Risks: excessive up_rate can induce undesirable oscillations, where transient local instabilities create global overreactions.

- **Low up_rate:**
  - Dampens the effect of local irregularities; upward propagation becomes more *gradual and averaged* across time windows or multiple nodes.
  - Favors *systemic stability* and long-term coherence, but can render the system *sluggish* or unresponsive to novel developments.

**Recommended Usage**

- In **real-time, interactive contexts** (e.g., live music improvisation, gaming AI), a higher up_rate (e.g., 0.7–1.0 scale) is recommended for lively, immediate emotional adaptation.
- For **ambient systems**, narrative pacing, or scenarios where *emotional inertia* is desired (e.g., background soundtrack generators, meditative apps), lower up_rate values (0.2–0.5) provide greater control and composure.
- Best practice involves *dynamically adjusting* up_rate in response to system meta-state (monitoring variance and volatility) to balance reactivity versus stability[43dcd9a7-70db-4a1f-b0ae-981daa162054](https://github.com/IdeasCosmos/Cosmos_emotional-vector/blob/main/cosmos_architecture.py?citationMarker=43dcd9a7-70db-4a1f-b0ae-981daa162054 "2").

---

### Parameter 2: down_decay

**Role**

The **down_decay** parameter moderates the *attenuation and fading* of signals as they descend from high-level layers down to subordinate modules and nodes. This governs how “instructions” from the Universal, Supra-systemic, or Macro layers are realized—or diluted—at the micro-node level.

**Effects on System Dynamics**

- **High down_decay:**
  - System-wide directives (e.g., global calming, context resets) *dissipate quickly*, exerting only temporary or minimal influence on individual node states.
  - Encourages *node independence* and emergent behavior, possibly at the expense of top-down consistency.
  - Suitable in systems requiring local creativity, adaptation, or heterogeneity.

- **Low down_decay:**
  - Directives maintain their *strength and persistence*, effectively saturating micro-level outputs with the global agenda.
  - Supports *systemic discipline* and mood coherence, but risks overriding unique local characteristics and diminishing bottom-up richness.

**Recommended Usage**

- Use higher down_decay (0.6–1.0) in *explorative or generative systems* where diversity and innovation at the node level are prized.
- Use lower down_decay (0.1–0.4) in *regulatory or safety-critical contexts* where tight alignment is non-negotiable (e.g., emotionally-aware interface for sensitive applications).
- In advanced deployments, *adaptive decay curves* (nonlinear, state-dependent) are employed to blend control according to current system entropy[43dcd9a7-70db-4a1f-b0ae-981daa162054](https://github.com/IdeasCosmos/Cosmos_emotional-vector/blob/main/cosmos_architecture.py?citationMarker=43dcd9a7-70db-4a1f-b0ae-981daa162054 "2")[43dcd9a7-70db-4a1f-b0ae-981daa162054](https://docs.nvidia.com/cosmos/latest/introduction.html?citationMarker=43dcd9a7-70db-4a1f-b0ae-981daa162054 "1").

### Parameter 3: top-k Guard Slider

**Role**

The **top-k guard slider** is a strategic filter controlling *how many of the most extreme or salient emotional outputs are permitted to influence scoring decisions* at any given processing cycle. It functions analogously to *attention gating* or *saliency thresholding* in cognitive architectures.

**Effects on System Dynamics**

- **High top-k value (allowing more signals):**
  - System incorporates a *broader swath of emotional vectors* in its final computation, which can yield nuanced, averaged emotional judgments.
  - Can reduce the impact of outlier or “noisy” signals.
  - Risks: may *dilute* the force of particularly strong or critical affective events, especially if drowned out by many lesser signals.

- **Low top-k value (allowing fewer signals):**
  - Only the *most dominant (highest scoring) vectors* participate, leading to more focused, intense, or extreme outputs.
  - Favors *decisive shifts* (good for dramatic moments or priority escalation), but can sensitize the system to volatility or input spikes.

**Recommended Usage**

- For applications demanding *subtlety and balance* (e.g., crowd emotional analysis, multi-agent narrative generation), use a **higher top-k** (e.g., 10–20 on a pool of 30 nodes).
- In scenarios where *focus and crisp reactivity* is desired (e.g., alarm signaling, behavioral gating), use a **lower top-k** (1–3).
- Periodic *dynamic adjustment* (e.g., top-k scales with recent system variance) is advocated to blend nuance with sensitivity as context demands[43dcd9a7-70db-4a1f-b0ae-981daa162054](https://github.com/IdeasCosmos/Cosmos_emotional-vector/blob/main/cosmos_architecture.py?citationMarker=43dcd9a7-70db-4a1f-b0ae-981daa162054 "2")[43dcd9a7-70db-4a1f-b0ae-981daa162054](https://github.com/IdeasCosmos/COSMOS-System?citationMarker=43dcd9a7-70db-4a1f-b0ae-981daa162054 "3").

## Integrating the Parameters: Interdependence and Best Practices

The operation of up_rate, down_decay, and the top-k guard slider is intrinsically interdependent. For instance:
- Setting *both* a high up_rate and a low top-k value creates an emotionally “jumpy” system, where the most active nodes dictate rapid global change.
- Conversely, a low up_rate, high down_decay, and high top-k value will yield a *very stable, mellow system* more resistant to dramatic affective fluctuations.

It is also critical to calibrate parameters in light of both **domain requirements** (real-time interactivity vs. background regulation) and **system configuration** (number of nodes, depth of hierarchy, target expressive bandwidth). Many implementations expose these parameters as *dynamic controls* within admin dashboards or API endpoints, allowing for *on-the-fly adjustment* by composers, machine learning models, or adaptive scripts[43dcd9a7-70db-4a1f-b0ae-981daa162054](https://github.com/IdeasCosmos/COSMOS-System?citationMarker=43dcd9a7-70db-4a1f-b0ae-981daa162054 "3").

Advanced COSMOS deployments can even employ *meta-control loops* wherein the system monitors its own entropy and emotional variance, then adapts up_rate, down_decay, and top-k guard in real time for optimal balance between stability and expressivity.
